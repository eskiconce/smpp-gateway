# Billing Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Billing por segmento: tarifas por prefix (`rate_tables`/`rate_entries`), reserva de saldo al encolar, débito real al aceptar (`submit_sm_resp` OK), ledger auditado (`transactions` con saldo resultante) y autenticación por API key de tenant en el endpoint de clientes.

**Architecture:** La tarifa se resuelve en el proceso de entrada (pipeline) tras el ruteo: tabla activa del tenant, entrada con prefix más largo (vigencia y conector opcionales), precio × segmentos. Se escribe `messages.amount`, se verifica saldo (prepaid) y se encola. La **reserva es un chequeo de suficiencia** (sin movimiento de saldo); el **débito real y único ocurre al aceptar** en el worker del conector, de forma atómica (saldo + fila `transactions` con saldo resultante, idempotente por `message_id`). Un subtipo prepaid/postpaid en el tenant (columna `mode`) decide si el saldo negativo es admisible. La API de clientes pasa a exigir `Authorization: Bearer <api_key>` del tenant.

**Tech Stack:** Go 1.22+, PostgreSQL 14+ (pgxpool, transacciones `BEGIN`/`UPDATE ... FOR UPDATE`), Redis 7+ (cola; el monto viaja en `queue.Item`), sin dependencias nuevas (stdlib `strings`, `math`, `errors`).

**Spec:** `docs/superpowers/specs/2026-09-12-smpp-gateway-design.md` — secciones "Billing" (débito al aceptar, tarifa por prefix_msisdn según conector, precio por segmento, reserva antes de encolar, saldo insuficiente → rechazo), "Modelo de datos" (tables `rate_tables`/`rate_entries`/`transactions`), "API REST" (`/api/v1/admin/{rate-tables,tenants}`, API Key por tenant) y "Decisiones clave" (prepaid/postpaid).

## Global Constraints

- Un solo binario `smppgw`; roles `server` (API + pipeline + reconciler) y `connector <id>` (consume su stream, mantiene bind outbound).
- Go 1.22+, PostgreSQL 14+, Redis 7+. Sin Docker.
- Solo SMPP 3.4; `registered_delivery=1` siempre; se persiste `smsc_msgid` en cuanto llega el `submit_sm_resp`.
- Postgres es fuente de verdad; Redis solo efímero (cola/caché). Todo lo que deba sobrevivir va a PG.
- **Billing:** débito al aceptar (no al encolar); si el proveedor rechaza no se cobra. Un mensaje se cobra una sola vez (índice único parcial).
- Ciclo de vida: `buffered → accepted → (frozen/resent) → final (DELIVRD/UNDELIV/EXPIRED/REJECTED)`.
- Dinero en `NUMERIC(14,4)`; en Go `float64` con redondeo `round4` (consistente con el helper `amtFloat` de M1/M2).
- TDD por paquete: test que falla → implementación mínima → test que pasa → commit.
- Sin comentarios en código salvo documentar invariantes. Commits en inglés (`feat:`/`docs:`). Nombres y copy en español (estados: `delivered/undeliv/expired/rejected`; transacciones `debit/credit`).
- Sin dependencias nuevas fuera de las ya usadas por M1/M2/M3 (go-redis v9, miniredis v2, pgxpool, x/time/rate).

## Plan base previo (estado tras M1, M2 y M3)

const tenantCols = `id, name, status, coalesce(routing_tag,''), balance, mode, api_key, created_at`

func scanTenants(rows pgx.Rows) ([]Tenant, error) {
    defer rows.Close()
    var out []Tenant
    for rows.Next() {
        var t Tenant
        rt := ""
        if err := rows.Scan(&t.ID, &t.Name, &t.Status, &rt, &t.Balance, &t.Mode, &t.ApiKey, &t.CreatedAt); err != nil {
            return nil, err
        }
        t.RoutingTag = rt
        out = append(out, t)
    }
    return out, rows.Err()
}

const rateTableCols = `id, tenant_id, name, active, created_at`

const rateEntryCols = `id, table_id, prefix, price, COALESCE(connector_id, 0), valid_from, valid_to`

// internal/store/pg.go — TenantRepo
func (r *PGRepo) ListTenants(ctx context.Context) ([]Tenant, error) {
    rows, err := r.pool.Query(ctx, `SELECT `+tenantCols+` FROM tenants ORDER BY created_at DESC`)
    if err != nil {
        return nil, err
    }
    return scanTenants(rows)
}

func (r *PGRepo) GetTenant(ctx context.Context, id string) (*Tenant, error) {
    row := r.pool.QueryRow(ctx, `SELECT `+tenantCols+` FROM tenants WHERE id=$1`, id)
    t, err := scanTenant(row)
    if errors.Is(err, pgx.ErrNoRows) {
        return nil, ErrNotFound
    }
    return t, err
}

func (r *PGRepo) GetTenantByAPIKey(ctx context.Context, apiKey string) (*Tenant, error) {
    row := r.pool.QueryRow(ctx, `SELECT `+tenantCols+` FROM tenants WHERE api_key=$1`, apiKey)
    t, err := scanTenant(row)
    if errors.Is(err, pgx.ErrNoRows) {
        return nil, ErrNotFound
    }
    return t, err
}

func scanTenant(row pgx.Row) (*Tenant, error) {
    var t Tenant
    rt := ""
    if err := row.Scan(&t.ID, &t.Name, &t.Status, &rt, &t.Balance, &t.Mode, &t.ApiKey, &t.CreatedAt); err != nil {
        return nil, err
    }
    t.RoutingTag = rt
    return &t, nil
}

func (r *PGRepo) CreateTenant(ctx context.Context, t *Tenant) error {
    if t.Mode == "" {
        t.Mode = "prepaid"
    }
    if t.Status == "" {
        t.Status = "active"
    }
    _, err := r.pool.Exec(ctx, `INSERT INTO tenants (id, name, status, routing_tag, balance, mode, api_key)
        VALUES ($1, $2, $3, $4, $5, $6, $7)`,
        t.ID, t.Name, t.Status, t.RoutingTag, t.Balance, t.Mode, t.ApiKey)
    return err
}

        var t RateTable

func (r *PGRepo) CreateRateTable(ctx context.Context, t *RateTable) error {
    return r.pool.QueryRow(ctx, `INSERT INTO rate_tables (tenant_id, name, active)

}

// internal/store/pg.go — LedgerRepo (transacción atómica; el débito es único por message_id)
func (r *PGRepo) Debit(ctx context.Context, tenantID, messageID string, amount float64) (float64, error) {
    tx, err := r.pool.Begin(ctx)
    if err != nil {
        return 0, err
    }
    defer tx.Rollback(ctx)

    var already int64
    if err := tx.QueryRow(ctx, `SELECT count(*) FROM transactions
        WHERE message_id=$1 AND type='debit'`, messageID).Scan(&already); err != nil {
        return 0, err
    }
    if already > 0 {
        // ya cobrado: devolver saldo actual sin registrar nada
        var bal float64
        if err := tx.QueryRow(ctx, `SELECT balance FROM tenants WHERE id=$1`, tenantID).Scan(&bal); err != nil {
            return 0, err
        }
        return bal, nil
    }

    var newBal float64
    if err := tx.QueryRow(ctx, `UPDATE tenants SET balance = balance - $2
        WHERE id=$1 RETURNING balance`, tenantID, amount).Scan(&newBal); err != nil {
        return 0, err
    }

// internal/billing/service.go
package billing

import (
    "context"
    "errors"
    "math"
    "strings"
    "time"

    "github.com/eskiconce/smpp-gateway/internal/store"
)

var (
    ErrNoRate             = errors.New("billing: sin tarifa para el destino")
    ErrInsufficientBalance = errors.New("billing: saldo insuficiente")
)

type Service struct {
    rates   store.RateRepo
    tenants store.TenantRepo
    ledger  store.LedgerRepo
}

func New(rates store.RateRepo, tenants store.TenantRepo, ledger store.LedgerRepo) *Service {
    return &Service{rates: rates, tenants: tenants, ledger: ledger}
}

// Price resuelve la tarifa vigente de mayor prefijo para el conector dado y la multiplica por segmentos.
func (s *Service) Price(ctx context.Context, tenantID string, connectorID int, msisdn string, segments int) (float64, error) {
    table, err := s.rates.GetActiveRateTable(ctx, tenantID)
    if err != nil {
        return 0, err
    }
    if table == nil {
        return 0, ErrNoRate
    }
    entries, err := s.rates.ListRateEntries(ctx, table.ID)
    if err != nil {
        return 0, err
    }
    now := time.Now()
    bestPrice, bestLen := 0.0, -1
    for _, e := range entries {
        if e.ConnectorID != 0 && e.ConnectorID != connectorID {
            continue
        }
        if e.ValidFrom != nil && now.Before(*e.ValidFrom) {
            continue
        }
        if e.ValidTo != nil && now.After(*e.ValidTo) {
            continue
        }
        if !strings.HasPrefix(msisdn, e.Prefix) {
            continue
        }
        if len(e.Prefix) > bestLen {
            bestPrice, bestLen = e.Price, len(e.Prefix)
        }
    }
    if bestLen < 0 {
        return 0, ErrNoRate
    }
    return round4(bestPrice * float64(segments)), nil
}

// Reserve verifica suficiencia: prepaid exige balance >= amount; postpaid siempre pasa.
func (s *Service) Reserve(ctx context.Context, tenantID string, amount float64) error {
    t, err := s.tenants.GetTenant(ctx, tenantID)
    if err != nil {
        return err
    }
    if t.Mode == "prepaid" && t.Balance < amount {
        return ErrInsufficientBalance
    }
    return nil
}

func (s *Service) Debit(ctx context.Context, tenantID, messageID string, amount float64) (float64, error) {
    return s.ledger.Debit(ctx, tenantID, messageID, amount)
}

func (s *Service) Credit(ctx context.Context, tenantID string, amount float64) (float64, error) {
    return s.ledger.Credit(ctx, tenantID, amount)
}

func round4(v float64) float64 {
    return math.Round(v*10000) / 10000
}

// internal/worker/worker.go — OnSubmitResp (ROK)
func (w *Worker) OnSubmitResp(seq uint32, status smpp.CommandStatus, msgid string) {
    it, ok := w.pending[seq]
    if !ok {
        return
    }
    delete(w.pending, seq)
    ctx := context.Background()
    if status != smpp.ESME_ROK {
        w.repo.SetSmscMsgid(ctx, it.ID, msgid)
        w.fallback(it, false)
        return
    }
    w.repo.SetSmscMsgid(ctx, it.ID, msgid)
    if w.dlrProc != nil {
        _ = w.dlrProc.Register(ctx, msgid, it.ID)
    }
    if w.biller != nil && it.Amount > 0 {
        if _, err := w.biller.Debit(ctx, it.TenantID, it.ID, it.Amount); err != nil {
            w.log.Warn("debit fallido", "message_id", it.ID, "err", err)
        }
    }
}

func (s *Server) handleSubmit(w http.ResponseWriter, r *http.Request) {
    key := strings.TrimPrefix(r.Header.Get("Authorization"), "Bearer ")
    if key == "" {
        http.Error(w, "api key requerida", http.StatusUnauthorized)
        return
    }
    tenant, err := s.repo.GetTenantByAPIKey(r.Context(), key)
    if err != nil {
        http.Error(w, "api key invalida", http.StatusUnauthorized)
        return
    }
    var req submitReq
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "json invalido", http.StatusBadRequest)
        return
    }
    out := s.p.SubmitOutgoing(req) // helper que arma pipeline.Outgoing desde submitReq
    if out.TenantID == "" {
        out.TenantID = tenant.ID
    }
    id, segments, err := s.p.Submit(r.Context(), out)
    if err != nil {
        switch {
        case errors.Is(err, billing.ErrInsufficientBalance):
            http.Error(w, "saldo insuficiente", http.StatusPaymentRequired)
        case errors.Is(err, billing.ErrNoRate):
            http.Error(w, "sin tarifa para el destino", http.StatusUnprocessableEntity)
        default:
            http.Error(w, err.Error(), http.StatusInternalServerError)
        }
        return
    }
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(submitResp{MessageID: id, Segments: segments})
}

func runConnector(ctx context.Context, cfg config.Config, repo *store.PGRepo, q queue.Queue, connectorID int) error {
    smppCfg, err := connectorSmppConfig(repo, connectorID)
    if err != nil {
        return err
    }
    cache, err := dlr.NewRedisCache(cfg.RedisURL, cfg.DLRTTL)
    if err != nil {
        return err
    }
    defer cache.Close()
    dlrProc := dlr.NewProcessor(cache, repo, dlr.NewWebhookNotifier(repo, cfg.WebhookTimeout))
    bill := billing.New(repo, repo, repo)

    w := worker.NewWorker(q, repo, worker.WithDLR(dlrProc), worker.WithBiller(bill))
    sess := session.New(session.Config{
        ConnectorID: smppCfg.ID, Host: smppCfg.Host, Port: smppCfg.Port,
        SystemID: smppCfg.SystemID, Password: smppCfg.Password,
        SourceAddr: smppCfg.SourceAddr, MsgPerSecond: amtFloat(smppCfg.MaxMsgPerSec),
    }, w)
    w.SetSession(sess)
    if err := sess.Dial(ctx); err != nil {
        return err
    }
    defer sess.Close()
    for prio := 0; prio <= 2; prio++ {
        key := queue.Key(connectorID, prio)
        go func(p int) {
            _ = q.Consume(ctx, key, fmt.Sprintf("cn-%d-%d", connectorID, p), w.Handle)
        }(prio)
    }
    <-ctx.Done()
    return nil
}
